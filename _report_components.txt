// ===== 错误汇报 =====
function openRpt() {
  var page = location.pathname.split('/').pop() || 'index.html';
  document.getElementById('rptPage').textContent = page;
  document.getElementById('rptDesc').value = '';
  document.getElementById('rptContact').value = '';
  document.getElementById('rptForm').style.display = '';
  document.getElementById('rptOk').style.display = 'none';
  document.getElementById('rptOverlay').classList.add('show');
}
function closeRpt() {
  document.getElementById('rptOverlay').classList.remove('show');
}
function submitRpt() {
  var desc = document.getElementById('rptDesc').value.trim();
  if (!desc) { document.getElementById('rptDesc').focus(); return; }
  var btn = document.getElementById('rptBtn');
  btn.disabled = true; btn.textContent = '提交中...';
  var payload = {
    page: document.getElementById('rptPage').textContent,
    desc: desc,
    contact: document.getElementById('rptContact').value.trim(),
    time: new Date().toLocaleString('zh-CN', {hour12:false}),
    user_agent: navigator.userAgent
  };
  fetch('/api/report', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  }).then(function(r){ return r.json(); }).then(function(d) {
    btn.disabled = false; btn.textContent = '提交汇报';
    document.getElementById('rptForm').style.display = 'none';
    document.getElementById('rptOk').style.display = 'block';
    setTimeout(closeRpt, 1800);
  }).catch(function(e) {
    btn.disabled = false; btn.textContent = '提交汇报';
    alert('汇报失败，请重试：' + e);
  });
}
